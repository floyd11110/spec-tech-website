ST EMBEDDED EXAMPLE LIBRARY

All Current Validated Examples

Collected examples for the current STDigitalIO and STAnalogIO APIs. These examples use the normal Arduino setup() / loop() model.

Digital I/O examples

1. Basic digital input controlling LED

#include <STDigitalIO.h>

STDigitalOutput led(13);
STDigitalInput button(2, INPUT_PULLUP);

void setup()
{
    led.begin(false);
    button.begin();
}

void loop()
{
    if (button.isLow())
    {
        led.on();
    }
    else
    {
        led.off();
    }
}

2. Output write using boolean values

#include <STDigitalIO.h>

STDigitalOutput led(13);

void setup()
{
    led.begin(false);
}

void loop()
{
    led.write(true);
    delay(1000);

    led.write(false);
    delay(1000);
}

3. Toggle output continuously

4. Raw input read

#include <STDigitalIO.h>

STDigitalOutput led(13);
STDigitalInput button(2, INPUT_PULLUP);

void setup()
{
    led.begin(false);
    button.begin();
}

void loop()
{
    int state = button.read();

    if (state)
    {
        led.on();
    }
    else
    {
        led.off();
    }
}

Because this uses INPUT_PULLUP, released is HIGH and pressed is LOW.

5. Debounced one-toggle-per-press

#include <STDigitalIO.h>

STDigitalOutput led(13);
STDigitalInput button(2, INPUT_PULLUP, 30);

void setup()
{
    led.begin(false);
    button.begin();
}

void loop()
{
    button.update();

    if (button.fell())
    {
        led.toggle();
    }
}

6. Separate press and release events

#include <STDigitalIO.h>

STDigitalInput button(2, INPUT_PULLUP, 30);

void setup()
{
    Serial.begin(115200);
    button.begin();
}

void loop()
{
    button.update();

    if (button.fell())
    {
        Serial.println("PRESSED / FELL");
    }

    if (button.rose())
    {
        Serial.println("RELEASED / ROSE");
    }
}

7. Stable-state regression example

#include <STDigitalIO.h>

STDigitalInput button(2, INPUT_PULLUP, 30);

void setup()
{
    Serial.begin(115200);
    button.begin();

    Serial.print("debounceMs() = ");
    Serial.println(button.debounceMs());

    button.setDebounce(50);

    Serial.print("after setDebounce(50) = ");
    Serial.println(button.debounceMs());
}

void loop()
{
    button.update();

    if (button.changed())
    {
        Serial.print("stableRead() = ");
        Serial.println(button.stableRead());

        Serial.print("stableHigh() = ");
        Serial.println(button.stableHigh() ? "TRUE" : "FALSE");

        Serial.print("stableLow() = ");
        Serial.println(button.stableLow() ? "TRUE" : "FALSE");

        Serial.print("rose() = ");
        Serial.println(button.rose() ? "TRUE" : "FALSE");

        Serial.print("fell() = ");
        Serial.println(button.fell() ? "TRUE" : "FALSE");
    }
}

9. Raw plus 10-sample average

#include <STAnalogIO.h>

STAnalogInput sensor(A0);

void setup()
{
    Serial.begin(115200);
    sensor.begin();
}

void loop()
{
    int raw = sensor.read();
    uint16_t avg = sensor.readAverage(10);

    Serial.print("RAW = ");
    Serial.print(raw);
    Serial.print(" | AVG10 = ");
    Serial.println(avg);

    delay(500);
}

11. Complete analog input monitor

#include <STAnalogIO.h>

STAnalogInput sensor(A0, 5.0f);

void setup()
{
    Serial.begin(115200);
    sensor.begin();
}

void loop()
{
    int raw = sensor.read();
    uint16_t average = sensor.readAverage(10);
    float voltage = sensor.readVoltage();
    float averageVoltage = sensor.readAverageVoltage(10);

    Serial.print("RAW = ");
    Serial.print(raw);
    Serial.print(" | AVG10 = ");
    Serial.print(average);
    Serial.print(" | V = ");
    Serial.print(voltage, 3);
    Serial.print(" | AVG V = ");
    Serial.println(averageVoltage, 3);

    delay(500);
}

12. Change reference voltage

#include <STAnalogIO.h>

STAnalogInput sensor(A0, 5.0f);

void setup()
{
    Serial.begin(115200);
    sensor.begin();

    Serial.print("PIN = ");
    Serial.println(sensor.pin());

    Serial.print("INITIAL VREF = ");
    Serial.println(sensor.referenceVoltage(), 3);

    sensor.setReferenceVoltage(4.900f);

    Serial.print("NEW VREF = ");
    Serial.println(sensor.referenceVoltage(), 3);

    Serial.print("ADC = ");
    Serial.println(sensor.read());

    Serial.print("VOLTAGE USING 4.900 VREF = ");
    Serial.println(sensor.readVoltage(), 3);
}

void loop()
{
}

Combined digital + analog examples

15. Potentiometer controls PWM brightness

#include <STAnalogIO.h>

STAnalogInput pot(A0);
STAnalogOutput pwm(9);

void setup()
{
    pot.begin();
    pwm.begin();
}

void loop()
{
    int raw = pot.readAverage(10);

    int pwmValue = map(raw, 0, 1023, 0, 255);

    pwm.write(pwmValue);

    delay(20);
}

This combines the validated analog averaging input with the validated PWM output. The Arduino map() function converts the 0–1023 ADC range to the 0–255 PWM range.

16. Button enables potentiometer-controlled PWM

#include <STDigitalIO.h>
#include <STAnalogIO.h>

STDigitalInput button(2, INPUT_PULLUP, 30);
STAnalogInput pot(A0);
STAnalogOutput pwm(9);

void setup()
{
    button.begin();
    pot.begin();
    pwm.begin();
}

void loop()
{
    button.update();

    if (button.stableLow())
    {
        int raw = pot.readAverage(10);
        int pwmValue = map(raw, 0, 1023, 0, 255);
        pwm.write(pwmValue);
    }
    else
    {
        pwm.off();
    }
}

17. Press toggles analog-output enable state

#include <STDigitalIO.h>
#include <STAnalogIO.h>

STDigitalInput button(2, INPUT_PULLUP, 30);
STAnalogInput pot(A0);
STAnalogOutput pwm(9);

bool enabled = false;

void setup()
{
    button.begin();
    pot.begin();
    pwm.begin();
}

void loop()
{
    button.update();

    if (button.fell())
    {
        enabled = !enabled;
    }

    if (enabled)
    {
        int raw = pot.readAverage(10);
        pwm.write(map(raw, 0, 1023, 0, 255));
    }
    else
    {
        pwm.off();
    }
}
