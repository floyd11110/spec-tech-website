ST EMBEDDED EXAMPLE LIBRARY

All Current Validated Examples

Collected examples for the current STDigitalIO, STAnalogIO, STTiming, and STPulse APIs. These examples use the normal Arduino setup() / loop() model.

Digital I/O examples

1. Basic digital input controlling LED

#include <STDigitalIO.h>

STDigitalOutput led(13);
STDigitalInput button(2, INPUT_PULLUP);

void setup()
{
    led.begin(false);
    button.begin();
}

void loop()
{
    if (button.isLow())
    {
        led.on();
    }
    else
    {
        led.off();
    }
}

2. Output write using boolean values

#include <STDigitalIO.h>

STDigitalOutput led(13);

void setup()
{
    led.begin(false);
}

void loop()
{
    led.write(true);
    delay(1000);

    led.write(false);
    delay(1000);
}

4. Raw input read

#include <STDigitalIO.h>

STDigitalOutput led(13);
STDigitalInput button(2, INPUT_PULLUP);

void setup()
{
    led.begin(false);
    button.begin();
}

void loop()
{
    int state = button.read();

    if (state)
    {
        led.on();
    }
    else
    {
        led.off();
    }
}

Because this uses INPUT_PULLUP, released is HIGH and pressed is LOW.

5. Debounced one-toggle-per-press

#include <STDigitalIO.h>

STDigitalOutput led(13);
STDigitalInput button(2, INPUT_PULLUP, 30);

void setup()
{
    led.begin(false);
    button.begin();
}

void loop()
{
    button.update();

    if (button.fell())
    {
        led.toggle();
    }
}

6. Separate press and release events

#include <STDigitalIO.h>

STDigitalInput button(2, INPUT_PULLUP, 30);

void setup()
{
    Serial.begin(115200);
    button.begin();
}

void loop()
{
    button.update();

    if (button.fell())
    {
        Serial.println("PRESSED / FELL");
    }

    if (button.rose())
    {
        Serial.println("RELEASED / ROSE");
    }
}

7. Stable-state regression example

#include <STDigitalIO.h>

STDigitalInput button(2, INPUT_PULLUP, 30);

void setup()
{
    Serial.begin(115200);
    button.begin();

    Serial.print("debounceMs() = ");
    Serial.println(button.debounceMs());

    button.setDebounce(50);

    Serial.print("after setDebounce(50) = ");
    Serial.println(button.debounceMs());
}

void loop()
{
    button.update();

    if (button.changed())
    {
        Serial.print("stableRead() = ");
        Serial.println(button.stableRead());

        Serial.print("stableHigh() = ");
        Serial.println(button.stableHigh() ? "TRUE" : "FALSE");

        Serial.print("stableLow() = ");
        Serial.println(button.stableLow() ? "TRUE" : "FALSE");

        Serial.print("rose() = ");
        Serial.println(button.rose() ? "TRUE" : "FALSE");

        Serial.print("fell() = ");
        Serial.println(button.fell() ? "TRUE" : "FALSE");
    }
}

11. Complete analog input monitor

#include <STAnalogIO.h>

{
    Serial.begin(115200);
    sensor.begin();
}

void loop()
{
    int raw = sensor.read();
    uint16_t average = sensor.readAverage(10);

    Serial.print(" | AVG10 = ");
    Serial.print(average);
    Serial.print(" | V = ");
    Serial.print(voltage, 3);
    Serial.print(" | AVG V = ");
    Serial.println(averageVoltage, 3);

    delay(500);
}

Combined digital + analog examples

15. Potentiometer controls PWM brightness

#include <STAnalogIO.h>

STAnalogInput pot(A0);
STAnalogOutput pwm(9);

void setup()
{
    pot.begin();
    pwm.begin();
}

void loop()
{
    int raw = pot.readAverage(10);

    int pwmValue = map(raw, 0, 1023, 0, 255);

    pwm.write(pwmValue);

    delay(20);
}

This combines the validated analog averaging input with the validated PWM output. The Arduino map() function converts the 0–1023 ADC range to the 0–255 PWM range.

16. Button enables potentiometer-controlled PWM

#include <STDigitalIO.h>
#include <STAnalogIO.h>

STDigitalInput button(2, INPUT_PULLUP, 30);
STAnalogInput pot(A0);
STAnalogOutput pwm(9);

void setup()
{
    button.begin();
    pot.begin();
    pwm.begin();
}

void loop()
{
    button.update();

    if (button.stableLow())
    {
        int raw = pot.readAverage(10);
        int pwmValue = map(raw, 0, 1023, 0, 255);
        pwm.write(pwmValue);
    }
    else
    {
        pwm.off();
    }
}

17. Press toggles analog-output enable state

#include <STDigitalIO.h>
#include <STAnalogIO.h>

STDigitalInput button(2, INPUT_PULLUP, 30);
STAnalogInput pot(A0);
STAnalogOutput pwm(9);

bool enabled = false;

void setup()
{
    button.begin();
    pot.begin();
    pwm.begin();
}

void loop()
{
    button.update();

    if (button.fell())
    {
        enabled = !enabled;
    }

    if (enabled)
    {
        int raw = pot.readAverage(10);
        pwm.write(map(raw, 0, 1023, 0, 255));
    }
    else
    {
        pwm.off();
    }
}
Validation note for combined examples

The individual STDigitalIO and STAnalogIO functions used here are hardware validated. These exact combined application sketches are documentation examples unless separately recorded as full application-level regression tests.

Timing examples

18. Basic non-blocking interval

#include <STTiming.h>

STTimer timer(1000);

void setup()
{
    Serial.begin(115200);
    timer.start();
}

void loop()
{
    if (timer.expired())
    {
        Serial.println("1 second elapsed");
        timer.restart();
    }
}

19. Elapsed and remaining time monitor

#include <STTiming.h>

STTimer timer(1000);

void setup()
{
    Serial.begin(115200);
    timer.start();
}

void loop()
{
    Serial.print("elapsed = ");
    Serial.print(timer.elapsed());

    Serial.print(" | remaining = ");
    Serial.println(timer.remaining());

    if (timer.expired())
    {
        timer.restart();
    }

    delay(250);
}

20. Reset, interval change and restart

#include <STTiming.h>

STTimer timer(1000);

void setup()
{
    Serial.begin(115200);

    timer.start();
    delay(400);

    timer.reset();          // elapsed = 0, stopped
    timer.setInterval(2000);
    timer.restart();        // elapsed = 0, running
}

void loop()
{
    if (timer.expired())
    {
        Serial.println("2000 ms expired");
    }
}

Pulse measurement examples

21. D9 to D7 pulse measurement

#include <STPulse.h>

STPulseInput pulse(7, 50000UL);

void setup()
{
    Serial.begin(115200);

    pulse.begin();
    tone(9, 100);
}

void loop()
{
    Serial.print("HIGH = ");
    Serial.print(pulse.highWidth());

    Serial.print(" | LOW = ");
    Serial.print(pulse.lowWidth());

    Serial.print(" | PERIOD = ");
    Serial.print(pulse.periodMicros());

    Serial.print(" | FREQ = ");
    Serial.print(pulse.frequencyHz(), 2);

    Serial.print(" | DUTY = ");
    Serial.print(pulse.dutyCycle(), 1);

    Serial.print(" | RPM = ");
    Serial.println(pulse.rpm(1), 1);

    delay(500);
}

22. RPM with pulses-per-revolution

float rpm1  = pulse.rpm(1);
float rpm2  = pulse.rpm(2);
float rpm20 = pulse.rpm(20);

At about 100 Hz, the validated results were approximately 6051 RPM @ 1 PPR, 3026 RPM @ 2 PPR, and 303 RPM @ 20 PPR.

23. Timeout / no-signal test

pulse.setTimeout(20000UL);

noTone(9);

unsigned long highUs = pulse.highWidth();
unsigned long lowUs = pulse.lowWidth();
unsigned long periodUs = pulse.periodMicros();
float hz = pulse.frequencyHz();

With the D9 source stopped, the validated result was HIGH = 0, LOW = 0, PERIOD = 0, FREQ = 0.00.
